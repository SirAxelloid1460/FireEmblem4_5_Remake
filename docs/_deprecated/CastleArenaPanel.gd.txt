extends Panel
class_name CastleArenaPanel

@onready var opponent_label: Label = $VBoxContainer/OpponentLabel
@onready var bet_amount_label: Label = $VBoxContainer/BetAmountLabel
@onready var preview_label: RichTextLabel = $VBoxContainer/PreviewLabel
@onready var fight_button: Button = $VBoxContainer/FightButton
@onready var leave_button: Button = $VBoxContainer/LeaveButton

var current_unit: Unit = null
var player_gold: int = 0
var opponent_unit: Unit = null
var bet_amount: int = 0

signal arena_finished(victory: bool, gold_earned: int, exp_gained: int)

func _ready():
	fight_button.pressed.connect(_on_fight_pressed)
	leave_button.pressed.connect(_on_leave_pressed)
	hide()

func start_arena(unit: Unit, gold: int):
	"""Inicia la arena con la unidad seleccionada"""
	current_unit = unit
	player_gold = gold
	
	# Generar oponente
	generate_opponent()
	
	# Calcular apuesta
	calculate_bet()
	
	# Mostrar preview
	show_battle_preview()
	
	show()

func generate_opponent():
	"""Genera un oponente basado en el nivel de la unidad"""
	opponent_unit = Unit.new()
	opponent_unit.unit_name = "Rival de Arena"
	opponent_unit.unit_class = "Fighter"
	opponent_unit.level = current_unit.level + randi_range(-2, 2)
	opponent_unit.level = max(1, opponent_unit.level)
	
	# Stats basadas en nivel
	opponent_unit.max_hp = 20 + (opponent_unit.level * 2)
	opponent_unit.current_hp = opponent_unit.max_hp
	opponent_unit.strength = 5 + opponent_unit.level
	opponent_unit.skill = 5 + opponent_unit.level
	opponent_unit.speed = 5 + opponent_unit.level
	opponent_unit.defense = 3 + (opponent_unit.level / 2)
	opponent_unit.resistance = 2 + (opponent_unit.level / 3)
	
	# Arma básica
	var weapon = Weapon.new()
	weapon.weapon_name = "Iron Sword"
	weapon.might = 5
	weapon.accuracy = 90
	opponent_unit.weapon = weapon

func calculate_bet():
	"""Calcula la apuesta de la arena"""
	# Basado en FE4: apuesta = 50 * nivel del oponente
	bet_amount = 50 * opponent_unit.level

func show_battle_preview():
	"""Muestra una preview del combate"""
	opponent_label.text = "%s - Lv %d %s" % [
		opponent_unit.unit_name,
		opponent_unit.level,
		opponent_unit.unit_class
	]
	
	bet_amount_label.text = "Apuesta: %d G" % bet_amount
	
	# Preview de stats
	var preview_text = "[b]%s[/b]\n" % current_unit.unit_name
	preview_text += "HP: %d  Atk: %d  Hit: %d\n\n" % [
		current_unit.current_hp,
		current_unit.calculate_attack(),
		current_unit.calculate_hit_rate(opponent_unit, 1)
	]
	
	preview_text += "VS\n\n"
	
	preview_text += "[b]%s[/b]\n" % opponent_unit.unit_name
	preview_text += "HP: %d  Atk: %d  Hit: %d" % [
		opponent_unit.current_hp,
		opponent_unit.calculate_attack(),
		opponent_unit.calculate_hit_rate(current_unit, 1)
	]
	
	preview_label.text = preview_text

func _on_fight_pressed():
	"""Inicia el combate de arena"""
	# Simular combate
	var result = simulate_arena_battle()
	
	if result["victory"]:
		var gold_earned = bet_amount * 2
		var exp_gained = opponent_unit.level * 10
		
		# Aplicar daño al jugador
		current_unit.current_hp = result["player_hp"]
		
		arena_finished.emit(true, gold_earned, exp_gained)
	else:
		# Derrota - el jugador muere
		current_unit.current_hp = 0
		arena_finished.emit(false, 0, 0)
	
	hide()

func simulate_arena_battle() -> Dictionary:
	"""Simula un combate de arena completo"""
	var player_hp = current_unit.current_hp
	var enemy_hp = opponent_unit.current_hp
	
	# Combate simplificado
	while player_hp > 0 and enemy_hp > 0:
		# Jugador ataca
		var player_hit = randf() * 100 < current_unit.calculate_hit_rate(opponent_unit, 1)
		if player_hit:
			var damage = max(0, current_unit.calculate_attack() - opponent_unit.defense)
			enemy_hp -= damage
		
		if enemy_hp <= 0:
			break
		
		# Enemigo ataca
		var enemy_hit = randf() * 100 < opponent_unit.calculate_hit_rate(current_unit, 1)
		if enemy_hit:
			var damage = max(0, opponent_unit.calculate_attack() - current_unit.defense)
			player_hp -= damage
	
	return {
		"victory": player_hp > 0,
		"player_hp": max(0, player_hp),
		"enemy_hp": max(0, enemy_hp)
	}

func _on_leave_pressed():
	"""Sale de la arena"""
	hide()
