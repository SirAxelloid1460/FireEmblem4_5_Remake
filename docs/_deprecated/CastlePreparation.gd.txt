func create_test_unit(unit_name: String, unit_class: String, level: int, gold: int) -> Unit:
	"""Crea una unidad de prueba"""
	var unit = Unit.new()
	unit.unit_name = unit_name
	unit.unit_class = unit_class
	unit.level = level
	unit.is_player_unit = true
	unit.max_hp = 20 + level * 2
	unit.current_hp = unit.max_hp
	unit.strength = 5 + level
	unit.skill = 5 + level
	unit.speed = 5 + level
	unit.defense = 3 + level / 2
	
	# Oro personal (FE4 feature)
	unit.personal_gold = gold
	
	return unit

func _on_location_selected(location: String):
	"""Maneja cuando se selecciona una ubicaciÃ³n del castillo"""
	if not selected_unit:
		show_info_popup("Aviso", "Selecciona una unidad primero")
		return
	
	current_location = location
	location_entered.emit(location)
	
	match location:
		"shop":
			open_shop()
		"blacksmith":
			open_blacksmith()
		"arena":
			open_arena()
		"fortune_teller":
			open_fortune_teller()
		"promotion":
			open_promotion()
		"convoy":
			open_convoy()

func open_shop():
	"""Abre la tienda"""
	var shop = ShopMenu.new()
	shop.setup(selected_unit, army_gold)
	shop.purchase_made.connect(_on_shop_purchase)
	ui_layer.add_child(shop)

func open_blacksmith():
	"""Abre el herrero"""
	var blacksmith = BlacksmithMenu.new()
	blacksmith.setup(selected_unit)
	blacksmith.repair_completed.connect(_on_weapon_repaired)
	ui_layer.add_child(blacksmith)

func open_arena():
	"""Abre la arena"""
	var arena = ArenaMenu.new()
	arena.setup(selected_unit)
	arena.battle_completed.connect(_on_arena_battle_completed)
	ui_layer.add_child(arena)

func open_fortune_teller():
	"""Abre la adivina (ve estadÃ­sticas y crecimiento)"""
	var fortune = FortuneTellerMenu.new()
	fortune.setup(selected_unit)
	ui_layer.add_child(fortune)

func open_promotion():
	"""Abre el menÃº de promociÃ³n"""
	var promotion = PromotionMenu.new()
	promotion.setup(selected_unit)
	promotion.promotion_completed.connect(_on_unit_promoted)
	ui_layer.add_child(promotion)

func open_convoy():
	"""Abre el convoy (inventario compartido)"""
	var convoy = ConvoyMenu.new()
	convoy.setup(selected_unit, convoy_items)
	convoy.items_changed.connect(_on_convoy_items_changed)
	ui_layer.add_child(convoy)

func _on_shop_purchase(item: Item, cost: int):
	"""Maneja cuando se compra algo en la tienda"""
	if selected_unit.personal_gold >= cost:
		selected_unit.personal_gold -= cost
		selected_unit.inventory.append(item)
		update_ui()
		show_info_popup("Compra exitosa", "%s comprÃ³ %s" % [selected_unit.unit_name, item.item_name])
	else:
		show_info_popup("Oro insuficiente", "No tienes suficiente oro")

func _on_weapon_repaired(weapon: Weapon, cost: int):
	"""Maneja cuando se repara un arma"""
	if selected_unit.personal_gold >= cost:
		selected_unit.personal_gold -= cost
		weapon.current_durability = weapon.max_durability
		update_ui()
		show_info_popup("ReparaciÃ³n exitosa", "%s fue reparado" % weapon.weapon_name)

func _on_arena_battle_completed(won: bool, gold_earned: int, exp_gained: int):
	"""Maneja el resultado de una batalla en la arena"""
	if won:
		selected_unit.personal_gold += gold_earned
		# TODO: Aplicar experiencia
		show_info_popup("Â¡Victoria en la Arena!", 
			"Ganaste %d G y %d EXP" % [gold_earned, exp_gained])
	else:
		show_info_popup("Derrota en la Arena", 
			"%s fue derrotado en la arena" % selected_unit.unit_name)
	
	update_ui()

func _on_unit_promoted(old_class: String, new_class: String):
	"""Maneja cuando una unidad es promocionada"""
	show_info_popup("Â¡PromociÃ³n!", 
		"%s ha sido promocionado de %s a %s!" % [selected_unit.unit_name, old_class, new_class])
	update_ui()

func _on_convoy_items_changed(items: Array):
	"""Maneja cambios en el convoy"""
	convoy_items = items
	update_ui()

func save_preparation_data():
	"""Guarda los datos de preparaciÃ³n"""
	var save_data = {
		"player_units": serialize_units(player_units),
		"army_gold": army_gold,
		"convoy_items": serialize_items(convoy_items),
		"current_chapter": current_chapter
	}
	
	# TODO: Integrar con sistema de guardado
	print("Datos guardados: ", save_data)

func serialize_units(units: Array[Unit]) -> Array:
	"""Serializa unidades para guardado"""
	var data = []
	for unit in units:
		data.append({
			"name": unit.unit_name,
			"class": unit.unit_class,
			"level": unit.level,
			"hp": unit.current_hp,
			"gold": unit.personal_gold
			# ... mÃ¡s datos
		})
	return data

func serialize_items(items: Array) -> Array:
	"""Serializa items para guardado"""
	var data = []
	for item in items:
		data.append({
			"name": item.item_name,
			"type": item.item_type
		})
	return data

func _on_start_battle():
	"""Inicia la batalla cuando el jugador estÃ¡ listo"""
	# ConfirmaciÃ³n
	show_confirmation_dialog(
		"Â¿Iniciar Batalla?",
		"Â¿EstÃ¡s listo para comenzar la batalla?",
		_confirm_start_battle

func _confirm_start_battle():
	"""Confirma y empieza la batalla"""
	preparation_complete = true
	preparation_completed.emit()
	
	# Guardar datos para la batalla
	save_preparation_data()
	
	# TransiciÃ³n a la batalla
	transition_to_battle()

func transition_to_battle():
	"""TransiciÃ³n a la escena de batalla"""
	# Fade out
	var fade = ColorRect.new()
	fade.color = Color.BLACK
	fade.modulate.a = 0
	fade.set_anchors_preset(Control.PRESET_FULL_RECT)
	ui_layer.add_child(fade)
	
	var tween = create_tween()
	tween.tween_property(fade, "modulate:a", 1.0, 1.0)
	await tween.finished
	
	# Cambiar a batalla
	get_tree().change_scene_to_file(next_battle_scene)
